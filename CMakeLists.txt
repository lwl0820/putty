cmake_minimum_required(VERSION 3.24)
project(PuTTYgen VERSION 0.1.0 LANGUAGES C CXX)
set(PUTTY_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(PG_BUILD_GUI "构建 Qt 图形界面" ON)
set(PUTTYGEN_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(PUTTYGEN_BINARY_DIR "${CMAKE_CURRENT_BINARY_DIR}")
include("${PUTTY_SOURCE_DIR}/cmake/UpstreamPuTTY.cmake")
add_library(pg-core STATIC src/core/pg_core.c src/core/pg_platform.c)
target_include_directories(pg-core PUBLIC src/core)
target_link_libraries(pg-core PRIVATE pg-putty)
if(WIN32)
  target_link_libraries(pg-core PRIVATE advapi32)
endif()
if(MSVC)
  target_compile_options(pg-core PRIVATE /W4 /utf-8)
else()
  target_compile_options(pg-core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests/core)
endif()
if(PG_BUILD_GUI)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets)
  if(BUILD_TESTING)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
  endif()
  add_subdirectory(src)
endif()
if(TARGET puttygen-gui)
  include(cmake/Packaging.cmake)
endif()
